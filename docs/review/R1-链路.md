## 1. P9 门控 deferred 名单断路（bug）

### 现象

import re, glob

def clean(p):
    p = p.strip(); out = []; i = 0
    while i < len(p):
        if p[i:i+2] == '${':
            depth = 0; j = i + 1
            while j < len(p):
                if p[j] == '{': depth += 1
                elif p[j] == '}':
                    depth -= 1
                    if depth == 0: break
                j += 1
            expr = p[i+2:j]; prev = out[-1] if out else ''
            if '?' in expr or expr.startswith('tq(') or 'toString()' in expr:
                pass
            elif '/' not in expr and j >= len(p)-1 and prev != '/':
                pass                      # 末尾 query 变量（如 ${params}）
            else:
                out.append('*')
            i = j + 1
        else:
            out.append(p[i]); i += 1
    p = ''.join(out).split('?')[0]
    p = re.sub(r'^/api', '', p)
    return tuple(s for s in p.split('/') if s)

# api.ts: method -> path
lines = open('web/src/lib/api.ts', encoding='utf-8').read().split('\n')
entries = []; cur = None
for i, l in enumerate(lines, 1):
    m = re.match(r'export const (api|p2api|p8api|benchapi) = \{', l)
    if m: cur = m.group(1); continue
    if cur:
        if l.startswith('};'): cur = None; continue
        mm = re.match(r'  ([A-Za-z0-9_]+)[:(]', l)

def extract(blob):
    mt = re.search(r'\b(get|post|put|patch|del|http)\s*[<(]', blob)
    fm = re.search(r'\bfetch\s*\(', blob)
    if fm and (not mt or fm.start() < mt.start()):

    elif mt:
        pos = mt.start()
        meth = {'get':'GET','post':'POST','put':'PUT','patch':'PATCH','del':'DELETE','http':None}[mt.group(1)]
    else:
        return None, None
    pm = re.search(r'[`\'"](/[^`\'"]*)', blob[pos:])
    p = pm.group(1) if pm else None
    if p and meth is None:
        hm = re.search(r'method:\s*"([A-Z]+)"', blob[pos:])
        meth = hm.group(1) if hm else 'GET'
    return meth, p

methods = {}

    if p: methods[(obj, name)] = (meth, p)

# 前端真正引用到的 api 方法（允许换行，如 api\n .getAgent）

# 后端路由

for f in glob.glob('server/*.go'):

    txt = open(f, encoding='utf-8').read()

    for l in txt.split('\n'):
